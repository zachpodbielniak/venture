# file and validates before saving -- and adjust the values. Everything an

# --------------------------------------------------------------------------
# Recurring expense: the classic subscription.
#
# Hosting bills on the 1st of every month whether you remember it or not;
# this makes the books agree. Cron format: minute hour day month weekday.
# --------------------------------------------------------------------------
pod monthly_hosting = cron->new("0 9 1 * *");
monthly_hosting->on_schedule => venture->create("expense",
	"description=VPS hosting", "vendor=Hosting Co", "amount=12.00",
	"category=software");

# --------------------------------------------------------------------------
# React to records changing: the venture module raises on_created,
# on_updated and on_deleted for every record write, from every surface --
# the web UI, the API, the CLI, the AI.
# --------------------------------------------------------------------------
pod sale_log = venture->new();
sale_log->on_created => log->write_info("A record was created");

# --------------------------------------------------------------------------
# A nightly count, to taste: point it at any registered type.
# --------------------------------------------------------------------------
pod nightly = cron->new("0 2 * * *");
nightly->on_schedule => venture->count("ticket");

# --------------------------------------------------------------------------
# Forge integration: react to a ticket being filed.
#
# The rules themselves live in records, not here -- which repository, which
# issue type, which runner, how far a run goes. This is only the trigger, so
# that a rule with a `manual` trigger can still be started by an automation
# when some other condition is met.
#
# Note what this does NOT do: the handler queues the run and returns. A pod
# handler that waited for a coding agent would sit inside podomation's
# dispatch for minutes, and that is not what the dispatch is for.
# --------------------------------------------------------------------------
pod ticket_watch = venture->new();
ticket_watch->on_created => log->write_info("A record was created; a forge rule may pick it up");
