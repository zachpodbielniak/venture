# --------------------------------------------------------------------------
# Recurring expense: the classic subscription.
#
# Hosting bills on the 1st of every month whether you remember it or not;
# this makes the books agree. Cron format: minute hour day month weekday.
# --------------------------------------------------------------------------
pod monthly_hosting = cron->new("0 9 1 * *");
monthly_hosting->on_schedule => venture->create("expense",
	"description=VPS hosting", "vendor=Hosting Co", "amount=12.00",
	"category=software");

# --------------------------------------------------------------------------
# React to records changing: the venture module raises on_created,
# on_updated and on_deleted for every record write, from every surface --
# the web UI, the API, the CLI, the AI.
# --------------------------------------------------------------------------
pod sale_log = venture->new();
sale_log->on_created => log->write_info("A record was created");

# --------------------------------------------------------------------------
# A nightly count, to taste: point it at any registered type.
# --------------------------------------------------------------------------
pod nightly = cron->new("0 2 * * *");
nightly->on_schedule => venture->count("ticket");
